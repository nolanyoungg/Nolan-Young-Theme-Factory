// Generated bundle.js

document.addEventListener('DOMContentLoaded', function() {
  const header = document.querySelector('header');
  const menuTriggers = document.querySelectorAll('.menu-trigger');
  const nolanMenu = document.querySelector('.nolan-menu');

  window.onscroll = () => {
    if (window.pageYOffset > 50) {
      header.classList.add('scrolled');
    } else {
      header.classList.remove('scrolled');
    }
  };

  menuTriggers.forEach(trigger => {
    trigger.addEventListener('click', function() {
      const targetMenuId = this.getAttribute('data-menu-id');
      if (nolanMenu.dataset.open === targetMenuId) {
        nolanMenu.classList.remove('open');
        nolanMenu.dataset.open = '';
      } else {
        nolanMenu.classList.add('open');
        nolanMenu.dataset.open = targetMenuId;
      }
    });
  });

  document.addEventListener('click', function(event) {
    if (!nolanMenu.contains(event.target) && !header.contains(event.target)) {
      nolanMenu.classList.remove('open');
      nolanMenu.dataset.open = '';
    }
  });

  document.addEventListener('keydown', function(event) {
    if (event.key === 'Escape') {
      nolanMenu.classList.remove('open');
      nolanMenu.dataset.open = '';
    }
  });
});
