// bundle.js
document.addEventListener('DOMContentLoaded', function() {
  const header = document.querySelector('.header');
  let lastScrollTop = 0;

  window.addEventListener('scroll', function() {
    if (window.pageYOffset > lastScrollTop) {
      header.classList.add('scrolled');
    } else {
      header.classList.remove('scrolled');
    }
    lastScrollTop = window.pageYOffset <= 0 ? 0 : window.pageYOffset;
  });

  const menuTriggers = document.querySelectorAll('.nav-item-button');
  const backdrop = document.createElement('div');
  backdrop.classList.add('backdrop');

  menuTriggers.forEach(trigger => {
    trigger.addEventListener('click', function() {
      if (this.classList.contains('active')) {
        this.classList.remove('active');
        backdrop.remove();
      } else {
        closeOtherMenus();
        this.classList.add('active');
        document.body.appendChild(backdrop);
        backdrop.addEventListener('click', closeAllMenus);
        document.addEventListener('keydown', escapeHandler);
      }
    });
  });

  function closeAllMenus() {
    menuTriggers.forEach(trigger => trigger.classList.remove('active'));
    backdrop.remove();
    document.removeEventListener('keydown', escapeHandler);
  }

  function closeOtherMenus() {
    menuTriggers.forEach(trigger => {
      if (trigger !== this && trigger.classList.contains('active')) {
        trigger.classList.remove('active');
      }
    });
  }

  function escapeHandler(event) {
    if (event.key === 'Escape') {
      closeAllMenus();
    }
  }

  backdrop.addEventListener('click', closeAllMenus);
});

// main.js
document.addEventListener('DOMContentLoaded', function() {
  const navItems = document.querySelectorAll('.nav-item');
  const headerHeight = document.querySelector('.header').clientHeight;

  navItems.forEach(item => {
    item.addEventListener('click', function() {
      if (item.classList.contains('has-menu')) {
        this.querySelector('.nolan-menu').classList.toggle('open');
      }
    });
  });

  const menuPanels = document.querySelectorAll('.nolan-menu');

  menuPanels.forEach(panel => {
    panel.addEventListener('mouseleave', function() {
      this.classList.remove('open');
    });
  });

  window.addEventListener('resize', () => {
    navItems.forEach(item => {
      item.style.setProperty('--header-height', `${headerHeight}px`);
    });
  });

  // Additional interactions can be added here
});
