document.addEventListener('DOMContentLoaded', function() {
  const header = document.querySelector('header');
  const logoLink = document.querySelector('.logo a');
  const navButtons = document.querySelectorAll('[data-menu-item]');
  const dropdownPanels = document.querySelectorAll('[data-menu-dropdown]');

  // Scroll event to toggle header state
  window.addEventListener('scroll', function() {
    if (window.scrollY > 0) {
      header.classList.add('scrolled');
    } else {
      header.classList.remove('scrolled');
    }
  });

  // Toggle dropdown panels
  navButtons.forEach(button => {
    button.addEventListener('click', function(event) {
      event.preventDefault();
      const panelId = this.getAttribute('aria-controls');
      const panel = document.getElementById(panelId);

      if (panel.classList.contains('open')) {
        panel.classList.remove('open');
        this.setAttribute('aria-expanded', 'false');
      } else {
        // Close all other panels
        dropdownPanels.forEach(p => p.classList.remove('open'));
        navButtons.forEach(b => b.setAttribute('aria-expanded', 'false'));

        panel.classList.add('open');
        this.setAttribute('aria-expanded', 'true');

        // Lock body scroll
        document.body.style.overflow = 'hidden';
      }
    });
  });

  // Close panel on clicking outside
  window.addEventListener('click', function(event) {
    if (!header.contains(event.target)) {
      dropdownPanels.forEach(panel => panel.classList.remove('open'));
      navButtons.forEach(button => button.setAttribute('aria-expanded', 'false'));

      // Unlock body scroll
      document.body.style.overflow = '';
    }
  });

  // Close panel on pressing Escape
  window.addEventListener('keydown', function(event) {
    if (event.key === 'Escape') {
      dropdownPanels.forEach(panel => panel.classList.remove('open'));
      navButtons.forEach(button => button.setAttribute('aria-expanded', 'false'));

      // Unlock body scroll
      document.body.style.overflow = '';
    }
  });

  // Handle left rail hover and focus in Services and About panels
  const leftRails = document.querySelectorAll('.left-rail');
  leftRails.forEach(rail => {
    rail.addEventListener('mouseover', function(event) {
      if (event.target.tagName === 'BUTTON') {
        updateRightContent(event.target);
      }
    });

    rail.addEventListener('focusin', function(event) {
      if (event.target.tagName === 'BUTTON') {
        updateRightContent(event.target);
      }
    });
  });
});

function updateRightContent(button) {
  const panelId = button.closest('.nolan-menu').getAttribute('data-menu-dropdown');
  const contentSections = document.querySelectorAll(`#${panelId} .content-section`);

  contentSections.forEach(section => {
    if (section.getAttribute('data-content-for') === button.getAttribute('data-menu-item')) {
      section.classList.add('active');
    } else {
      section.classList.remove('active');
    }
  });
}
