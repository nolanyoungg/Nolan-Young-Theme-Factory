function nolan_young_theme_process_contact_form() {
    if (isset($_POST['contact-form-nonce']) && wp_verify_nonce($_POST['contact-form-nonce'], 'contact_form_nonce') && isset($_POST['name']) && isset($_POST['email']) && isset($_POST['message'])) {

        echo '<div class="success-message">Thank you for your submission!</div>';
    }

<?php
// Handle newsletter signups
function nolan_young_theme_process_newsletter_signup() {
    if (isset($_POST['newsletter-signup-nonce']) && wp_verify_nonce($_POST['newsletter-signup-nonce'], 'newsletter_signup_nonce') && isset($_POST['email'])) {
        $email = sanitize_email($_POST['email']);
        $name = isset($_POST['name']) ? sanitize_text_field($_POST['name']) : '';

        if (empty($email)) {
            wp_die('Please enter a valid email address.');
        }

        // Normalize and store subscriber
        $subscriber = get_user_by('email', $email);
        if (!$subscriber) {
            $subscriber_id = wp_insert_user(array(
                'user_email' => $email,
                'first_name' => $name,
                'role' => 'subscriber',
            ));
            update_user_meta($subscriber_id, 'newsletter_status', 'Active');
        } elseif (get_user_meta($subscriber->ID, 'newsletter_status', true) === 'Unsubscribed') {
            update_user_meta($subscriber->ID, 'newsletter_status', 'Active');
        }

        // Display success message
        echo '<div class="success-message">Thank you for subscribing!</div>';
    }
}
add_action('init', 'nolan_young_theme_process_newsletter_signup');

// Unsubscribe functionality
function nolan_young_theme_process_unsubscribe() {
    if (isset($_GET['unsubscribe_token']) && isset($_GET['email'])) {
        $email = sanitize_email($_GET['email']);
        $token = sanitize_text_field($_GET['unsubscribe_token']);

        // Validate token and update status
        $subscriber = get_user_by('email', $email);
        if ($subscriber && wp_hash(get_userdata($subscriber->ID)->user_login) === $token) {
            update_user_meta($subscriber->ID, 'newsletter_status', 'Unsubscribed');
            echo '<div class="success-message">You have been unsubscribed successfully!</div>';
        } else {
            wp_die('Invalid unsubscribe token.');
        }
    }
}
add_action('init', 'nolan_young_theme_process_unsubscribe');
?>

// Sanitize and validate email
function nolan_young_theme_sanitize_email($email) {
    $sanitized = sanitize_email($email);
    if (is_email($sanitized)) {
        return $sanitized;
    }
    return '';
}

    return sanitize_text_field($text);

    if ($type === 'success') {
        echo '<div class="success-message">' . esc_html($message) . '</div>';
    } elseif ($type === 'error') {
        echo '<div class="error-message">' . esc_html($message) . '</div>';
    }
}
